const STORAGE_KEY = 'gps-destinations-v1'; const PROVIDER_KEY = 'gps-provider-v1'; const defaultDestinations = [ { id: crypto.randomUUID(), name: 'Maison', address: 'Votre adresse', icon: '🏠' }, { id: crypto.randomUUID(), name: 'Travail', address: 'Votre lieu de travail', icon: '💼' }, { id: crypto.randomUUID(), name: 'Courses', address: 'Votre centre commercial', icon: '🛒' }, { id: crypto.randomUUID(), name: 'Favori', address: 'Votre destination favorite', icon: '⭐' } ]; let destinations = loadDestinations(); let provider = localStorage.getItem(PROVIDER_KEY) || 'google'; let editingId = null; const grid = document.getElementById('destinationsGrid'); const template = document.getElementById('destinationTemplate'); const searchInput = document.getElementById('searchInput'); const countLabel = document.getElementById('countLabel'); const emptyState = document.getElementById('emptyState'); const addBtn = document.getElementById('addBtn'); const dialog = document.getElementById('destinationDialog'); const form = document.getElementById('destinationForm'); const closeDialogBtn = document.getElementById('closeDialogBtn'); const deleteBtn = document.getElementById('deleteBtn'); const destinationId = document.getElementById('destinationId'); const nameInput = document.getElementById('nameInput'); const addressInput = document.getElementById('addressInput'); const iconInput = document.getElementById('iconInput'); const dialogTitle = document.getElementById('dialogTitle'); function loadDestinations() { try { const saved = JSON.parse(localStorage.getItem(STORAGE_KEY)); return Array.isArray(saved) && saved.length ? saved : defaultDestinations; } catch { return defaultDestinations; } } function saveDestinations() { localStorage.setItem(STORAGE_KEY, JSON.stringify(destinations)); } function navigationUrl(address) { const q = encodeURIComponent(address.trim()); if (provider === 'waze') { return `https://www.waze.com/ul?q=${q}&navigate=yes`; } return `https://www.google.com/maps/dir/?api=1&destination=${q}&travelmode=driving`; } function openNavigation(address) { if (!address || /votre /i.test(address)) { editDestination(destinations.find(d => d.address === address)?.id); return; } // assign() est volontairement utilisé plutôt que window.open() : // sur iPhone, cela maximise les chances que le lien universel passe à l'app GPS. window.location.assign(navigationUrl(address)); } function render() { const query = searchInput.value.trim().toLocaleLowerCase('fr'); const filtered = destinations.filter(item => `${item.name} ${item.address}`.toLocaleLowerCase('fr').includes(query) ); grid.innerHTML = ''; filtered.forEach(item => { const node = template.content.cloneNode(true); const card = node.querySelector('.destination-card'); const openBtn = node.querySelector('.destination-open'); const editBtn = node.querySelector('.edit-btn'); node.querySelector('.destination-icon').textContent = item.icon || '📍'; node.querySelector('.destination-name').textContent = item.name; node.querySelector('.destination-address').textContent = item.address; openBtn.addEventListener('click', () => openNavigation(item.address)); editBtn.addEventListener('click', (event) => { event.stopPropagation(); editDestination(item.id); }); card.dataset.id = item.id; grid.appendChild(node); }); countLabel.textContent = `${filtered.length} destination${filtered.length > 1 ? 's' : ''}`; emptyState.hidden = filtered.length !== 0; } function openAddDialog() { editingId = null; destinationId.value = ''; dialogTitle.textContent = 'Ajouter'; nameInput.value = ''; addressInput.value = ''; iconInput.value = '📍'; deleteBtn.hidden = true; dialog.showModal(); setTimeout(() => nameInput.focus(), 50); } function editDestination(id) { const item = destinations.find(d => d.id === id); if (!item) return; editingId = id; destinationId.value = id; dialogTitle.textContent = 'Modifier'; nameInput.value = item.name; addressInput.value = item.address; iconInput.value = item.icon || '📍'; deleteBtn.hidden = false; dialog.showModal(); } function closeDialog() { dialog.close(); editingId = null; } form.addEventListener('submit', (event) => { event.preventDefault(); const name = nameInput.value.trim(); const address = addressInput.value.trim(); const icon = iconInput.value; if (!name || !address) return; if (editingId) { destinations = destinations.map(item => item.id === editingId ? { ...item, name, address, icon } : item ); } else { destinations.unshift({ id: crypto.randomUUID(), name, address, icon }); } saveDestinations(); closeDialog(); render(); }); deleteBtn.addEventListener('click', () => { if (!editingId) return; destinations = destinations.filter(item => item.id !== editingId); saveDestinations(); closeDialog(); render(); }); addBtn.addEventListener('click', openAddDialog); closeDialogBtn.addEventListener('click', closeDialog); searchInput.addEventListener('input', render); for (const button of document.querySelectorAll('.provider')) { button.classList.toggle('active', button.dataset.provider === provider); button.addEventListener('click', () => { provider = button.dataset.provider; localStorage.setItem(PROVIDER_KEY, provider); document.querySelectorAll('.provider').forEach(btn => btn.classList.toggle('active', btn === button) ); }); } if ('serviceWorker' in navigator) { window.addEventListener('load', () => { navigator.serviceWorker.register('./service-worker.js').catch(() => {}); }); } render();
